export const GroupByBody = () => (
  <>
    Group results by <code>genre</code> and count entries per group using{" "}
    <code>_count</code>. Any field can be used in <code>groupBy</code>.
  </>
);

export const basicCode = `{
  Books {
    title
    rating
    author {
      name
    }
  }
}`;

export const basicResult = `{
  "data": {
    "Books": [
      { "title": "Dune", "rating": 4.9, "author": { "name": "Frank Herbert" } },
      { "title": "Foundation", "rating": 4.7, "author": { "name": "Isaac Asimov" } },
      { "title": "Neuromancer", "rating": 4.5, "author": { "name": "William Gibson" } },
      { "title": "Snow Crash", "rating": 4.6, "author": { "name": "Neal Stephenson" } },
      { "title": "The Left Hand of Darkness", "rating": 4.4, "author": { "name": "Ursula K. Le Guin" } },
      { "title": "Hyperion", "rating": 4.8, "author": { "name": "Dan Simmons" } }
    ]
  }
}`;

export const filterCode = `{
  Books(filter: { rating: { _ge: 4.6 } }) {
    title
    rating
    author {
      name
    }
  }
}`;

export const filterResult = `{
  "data": {
    "Books": [
      { "title": "Dune", "rating": 4.9, "author": { "name": "Frank Herbert" } },
      { "title": "Foundation", "rating": 4.7, "author": { "name": "Isaac Asimov" } },
      { "title": "Snow Crash", "rating": 4.6, "author": { "name": "Neal Stephenson" } },
      { "title": "Hyperion", "rating": 4.8, "author": { "name": "Dan Simmons" } }
    ]
  }
}`;

export const groupByCode = `{
  Books(groupBy: [genre]) {
    genre
    _count
    _group {
      title
      rating
    }
  }
}`;
